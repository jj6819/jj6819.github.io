Below is a lightweight **animated “Sleep Screen”** you can drop into your vanilla JS app. It’s designed to be sleep-inducing: **dark, low-contrast**, **slow breathing orb**, subtle drifting “stars,” and a tiny noise texture. It also respects **prefers-reduced-motion** and lets users tap/press ESC to exit.

### What it looks/feels like

* Background: deep navy → near-black gradient (very low contrast)
* Center: softly blurred “orb” that **breathes** (slow expand/contract)
* Ambient: tiny particles drifting slowly (optional)
* Noise: faint film grain (helps reduce “digital harshness”)

---

## 1) Add this HTML anywhere (near the end of `<body>`)

```html
<!-- Sleep Screen Overlay -->
<div id="sleepScreen" aria-hidden="true">
  <div class="sleep-bg"></div>
  <canvas id="sleepStars" width="1" height="1"></canvas>

  <div class="sleep-orb" aria-hidden="true"></div>
  <div class="sleep-vignette" aria-hidden="true"></div>

  <!-- Minimal hint (fades out quickly) -->
  <div class="sleep-hint" aria-hidden="true">Tap to exit</div>
</div>
```

## 3) Add this JavaScript (vanilla)

```html
<script>
(() => {
  const sleepEl = document.getElementById("sleepScreen");
  const canvas = document.getElementById("sleepStars");
  const ctx = canvas.getContext("2d", { alpha: true });

  let running = false;
  let rafId = null;
  let stars = [];
  let lastTs = 0;

  const prefersReducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  function resizeCanvas() {
    const dpr = Math.min(window.devicePixelRatio || 1, 2); // cap DPR for perf
    canvas.width = Math.floor(window.innerWidth * dpr);
    canvas.height = Math.floor(window.innerHeight * dpr);
    canvas.style.width = "100%";
    canvas.style.height = "100%";
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  }

  function initStars() {
    stars = [];
    const w = window.innerWidth;
    const h = window.innerHeight;

    // Keep it subtle: fewer stars than you think
    const count = Math.round(Math.min(140, (w * h) / 16000)); // scales with screen size
    for (let i = 0; i < count; i++) {
      stars.push({
        x: Math.random() * w,
        y: Math.random() * h,
        r: Math.random() * 1.2 + 0.2,
        a: Math.random() * 0.35 + 0.05,
        vx: (Math.random() * 0.12 + 0.02) * (Math.random() < 0.5 ? -1 : 1),
        vy: (Math.random() * 0.08 + 0.01),
        tw: Math.random() * 0.6 + 0.2,      // twinkle speed factor
        ph: Math.random() * Math.PI * 2      // phase
      });
    }
  }

  function draw(ts) {
    if (!running) return;

    const w = window.innerWidth;
    const h = window.innerHeight;
    const dt = Math.min(0.05, (ts - lastTs) / 1000 || 0.016);
    lastTs = ts;

    ctx.clearRect(0, 0, w, h);

    // Very gentle motion + twinkle
    for (const s of stars) {
      s.x += s.vx * (dt * 60);
      s.y += s.vy * (dt * 60);

      // wrap
      if (s.x < -5) s.x = w + 5;
      if (s.x > w + 5) s.x = -5;
      if (s.y > h + 5) s.y = -5;

      const twinkle = 0.65 + 0.35 * Math.sin(ts * 0.0005 * s.tw + s.ph);
      const alpha = s.a * twinkle;

      ctx.beginPath();
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fillStyle = `rgba(210, 230, 255, ${alpha.toFixed(3)})`;
      ctx.fill();
    }

    rafId = requestAnimationFrame(draw);
  }

  function openSleepScreen() {
    sleepEl.classList.add("is-open");
    sleepEl.setAttribute("aria-hidden", "false");
    document.documentElement.style.overflow = "hidden";

    if (!prefersReducedMotion) {
      resizeCanvas();
      initStars();
      running = true;
      lastTs = performance.now();
      rafId = requestAnimationFrame(draw);
    }
  }

  function closeSleepScreen() {
    sleepEl.classList.remove("is-open");
    sleepEl.setAttribute("aria-hidden", "true");
    document.documentElement.style.overflow = "";

    running = false;
    if (rafId) cancelAnimationFrame(rafId);
    rafId = null;
  }

  // Close interactions: tap anywhere, ESC key
  sleepEl.addEventListener("click", closeSleepScreen);
  window.addEventListener("keydown", (e) => {
    if (e.key === "Escape" && sleepEl.classList.contains("is-open")) closeSleepScreen();
  });

  // Keep canvas correct on resize
  window.addEventListener("resize", () => {
    if (!sleepEl.classList.contains("is-open") || prefersReducedMotion) return;
    resizeCanvas();
    initStars();
  });

  // Expose simple API for your app
  window.NightOwlSleepScreen = { open: openSleepScreen, close: closeSleepScreen };
})();
</script>
```

## 4) Add a button in your app to trigger it

Anywhere in your existing UI:

```html
<button id="openSleepScreenBtn" type="button">Sleep Screen</button>
<script>
  document.getElementById("openSleepScreenBtn").addEventListener("click", () => {
    window.NightOwlSleepScreen.open();
  });
</script>
```
