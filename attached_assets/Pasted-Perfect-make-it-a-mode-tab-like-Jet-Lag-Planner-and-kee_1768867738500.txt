Perfect — make it a **mode tab** (like Jet Lag Planner), and keep it **pure visual** (no “tap to exit” text, no UI inside the scene). The key difference vs an overlay is: **start the animation only when the tab is active, stop it when you switch away** (so you don’t burn CPU in the background).

Below is a drop-in plan + code you can adapt to your existing tab system.

## 3) JS: start/stop animation when the tab is active

This is the crucial part for a “mode” implementation: don’t animate when the user isn’t on the Sleep Screen tab.

```html
<script>
(() => {
  const prefersReducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  const tabRoot = document.getElementById("sleepScreenTab");
  const canvas = document.getElementById("sleepStars");
  const ctx = canvas.getContext("2d", { alpha: true });

  let running = false;
  let rafId = null;
  let stars = [];
  let lastTs = 0;

  function resizeCanvas() {
    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = Math.floor(tabRoot.clientWidth * dpr);
    canvas.height = Math.floor(tabRoot.clientHeight * dpr);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  }

  function initStars() {
    stars = [];
    const w = tabRoot.clientWidth;
    const h = tabRoot.clientHeight;
    const count = Math.round(Math.min(140, (w * h) / 16000));

    for (let i = 0; i < count; i++) {
      stars.push({
        x: Math.random() * w,
        y: Math.random() * h,
        r: Math.random() * 1.2 + 0.2,
        a: Math.random() * 0.35 + 0.05,
        vx: (Math.random() * 0.12 + 0.02) * (Math.random() < 0.5 ? -1 : 1),
        vy: (Math.random() * 0.08 + 0.01),
        tw: Math.random() * 0.6 + 0.2,
        ph: Math.random() * Math.PI * 2
      });
    }
  }

  function draw(ts) {
    if (!running) return;

    const w = tabRoot.clientWidth;
    const h = tabRoot.clientHeight;
    const dt = Math.min(0.05, (ts - lastTs) / 1000 || 0.016);
    lastTs = ts;

    ctx.clearRect(0, 0, w, h);

    for (const s of stars) {
      s.x += s.vx * (dt * 60);
      s.y += s.vy * (dt * 60);

      if (s.x < -5) s.x = w + 5;
      if (s.x > w + 5) s.x = -5;
      if (s.y > h + 5) s.y = -5;

      const twinkle = 0.65 + 0.35 * Math.sin(ts * 0.0005 * s.tw + s.ph);
      const alpha = s.a * twinkle;

      ctx.beginPath();
      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
      ctx.fillStyle = `rgba(210, 230, 255, ${alpha.toFixed(3)})`;
      ctx.fill();
    }

    rafId = requestAnimationFrame(draw);
  }

  function startSleepScreen() {
    if (running || prefersReducedMotion) return;
    resizeCanvas();
    initStars();
    running = true;
    lastTs = performance.now();
    rafId = requestAnimationFrame(draw);
  }

  function stopSleepScreen() {
    running = false;
    if (rafId) cancelAnimationFrame(rafId);
    rafId = null;
  }

  // Hook into your tab switching:
  // Call window.NightOwlSleepScreenTab.onModeChange(activeMode)
  window.NightOwlSleepScreenTab = {
    onModeChange(activeMode) {
      if (activeMode === "sleepscreen") startSleepScreen();
      else stopSleepScreen();
    }
  };

  window.addEventListener("resize", () => {
    if (!running || prefersReducedMotion) return;
    resizeCanvas();
    initStars();
  });
})();
</script>
```

## 4) Tie it into your existing mode/tab switcher

Wherever you handle tab switching now, add one line after setting the active mode:

```js
function setMode(mode) {
  // your existing logic that shows/hides panels
  // ...

  // start/stop sleep screen rendering
  if (window.NightOwlSleepScreenTab) {
    window.NightOwlSleepScreenTab.onModeChange(mode);
  }
}
```

That’s it.

## 5) Small UX polish (optional but recommended)

* When Sleep Screen is active, consider **hiding other page controls** (or at least scrolling) so it feels calm.
* Keep the animation slow (12–16s breath cycle). If you want it even more sedating, bump `sleepBreath` to **16s**.
